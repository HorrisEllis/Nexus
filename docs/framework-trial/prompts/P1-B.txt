function decide(verdict, { alreadyDone = false } = {}) {

  if (verdict === 'passes-already' && alreadyDone) return 'accept-already'; // added later: a phase whose files already exist on disk can never get a failing test, so a passing test is kept as a regression test
  return 'reject';
}
